async function loadRealm() { try { // 1. Fetch assets map and config const [assetsRes, configRes] = await Promise.all([ fetch('./world.assets.yaml'), fetch('./world.config.yaml') ]); if (!assetsRes.ok || !configRes.ok) throw new Error("Could not fetch YAML configuration files"); const assetsData = jsyaml.load(await assetsRes.text()); const configData = jsyaml.load(await configRes.text()); // Spawn puts assets under an 'assets' dictionary key const assetRegistry = assetsData.assets || assetsData; console.log("Parsed Asset Registry:", assetRegistry); // 2. Locate the active place definition (e.g. 'main') const placeName = configData.defaultPlace || 'main'; let placeData = null; // Try fetching places/main.json, fallback to places/main.yaml let placeRes = await fetch(`./places/${placeName}.json`); if (placeRes.ok) { placeData = await placeRes.json(); } else { placeRes = await fetch(`./places/${placeName}.yaml`); if (placeRes.ok) { placeData = jsyaml.load(await placeRes.text()); } } console.log(`Place [${placeName}] Data:`, placeData); if (!placeData) { console.warn(`Could not find place file at ./places/${placeName}.json or .yaml`); return; } // 3. Extract items/entities from the place file const items = placeData.entities || placeData.objects || placeData.items || placeData.props || []; console.log(`Found ${items.length} place entities to spawn.`); items.forEach((item) => { // Find matching asset definition by ID or template const assetKey = item.assetId || item.asset || item.template || item.type; const assetDef = assetRegistry[assetKey]; const modelUrl = assetDef?.url || assetDef?.file || assetDef?.path || item.url; if (!modelUrl) return; gltfLoader.load( modelUrl, (gltf) => { const root = gltf.scene; // Position if (item.position) { const p = item.position; root.position.set( Array.isArray(p) ? p[0] : (p.x || 0), Array.isArray(p) ? p[1] : (p.y || 0), Array.isArray(p) ? p[2] : (p.z || 0) ); } // Rotation if (item.rotation) { const r = item.rotation; root.rotation.set( Array.isArray(r) ? r[0] : (r.x || 0), Array.isArray(r) ? r[1] : (r.y || 0), Array.isArray(r) ? r[2] : (r.z || 0) ); } // Scale if (item.scale !== undefined) { const s = item.scale; if (typeof s === 'number') { root.scale.set(s, s, s); } else { root.scale.set( Array.isArray(s) ? s[0] : (s.x || 1), Array.isArray(s) ? s[1] : (s.y || 1), Array.isArray(s) ? s[2] : (s.z || 1) ); } } scene.add(root); }, undefined, (err) => console.error(`Failed to load asset ${assetKey} from ${modelUrl}:`, err) ); }); } catch (err) { console.error("World loader error:", err); } }
